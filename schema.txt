create schema if not exists recipes;

create table recipes.cuisines (
    id uuid primary key default gen_random_uuid(),
    name text not null unique
);

insert into recipes.cuisines (name) values
    ('Italian'), ('Mexican'), ('Indian'), ('Chinese'), ('Thai'),
    ('French'), ('Japanese'), ('Mediterranean'), ('American'), ('Other');

create table recipes.recipes (
    id uuid primary key default gen_random_uuid(),
    user_id uuid not null references auth.users(id) on delete cascade,
    cuisine_id uuid references recipes.cuisines(id),
    title text not null,
    description text,
    instructions text,
    servings int,
    prep_time_minutes int,
    cook_time_minutes int,
    image_path text,
    created_at timestamptz not null default now()
);

create table recipes.ingredients (
    id uuid primary key default gen_random_uuid(),
    recipe_id uuid not null references recipes.recipes(id) on delete cascade,
    name text not null,
    quantity numeric,
    unit text,
    position int not null default 0
);

create table recipes.recipe_shares (
    id uuid primary key default gen_random_uuid(),
    recipe_id uuid not null references recipes.recipes(id) on delete cascade,
    shared_with_user_id uuid not null references auth.users(id) on delete cascade,
    unique (recipe_id, shared_with_user_id)
);

alter table recipes.cuisines enable row level security;
alter table recipes.recipes enable row level security;
alter table recipes.ingredients enable row level security;
alter table recipes.recipe_shares enable row level security;

create policy "view cuisines" on recipes.cuisines
    for select using (auth.role() = 'authenticated');

create policy "owners full access" on recipes.recipes
    for all using (auth.uid() = user_id);

create policy "shared users can view" on recipes.recipes
    for select using (
        exists (select 1 from recipes.recipe_shares s
                where s.recipe_id = id and s.shared_with_user_id = auth.uid())
    );

create policy "view ingredients if recipe accessible" on recipes.ingredients
    for select using (
        exists (select 1 from recipes.recipes r
                where r.id = recipe_id
                and (r.user_id = auth.uid()
                     or exists (select 1 from recipes.recipe_shares s
                                where s.recipe_id = r.id and s.shared_with_user_id = auth.uid())))
    );

create policy "owners manage ingredients" on recipes.ingredients
    for all using (
        exists (select 1 from recipes.recipes r where r.id = recipe_id and r.user_id = auth.uid())
    );

create policy "add self via share" on recipes.recipe_shares
    for insert with check (auth.uid() = shared_with_user_id);

create policy "view own shares" on recipes.recipe_shares
    for select using (auth.uid() = shared_with_user_id);
