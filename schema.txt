-- Create a dedicated schema so this stays separate from your other app's tables
create schema if not exists recipes;

-- Cuisines: shared reference data, one per recipe
create table recipes.cuisines (
    id uuid primary key default gen_random_uuid(),
    name text not null unique
);

insert into recipes.cuisines (name) values
    ('Italian'), ('Mexican'), ('Indian'), ('Chinese'), ('Thai'),
    ('French'), ('Japanese'), ('Mediterranean'), ('American'), ('Other');

-- Recipes
create table recipes.recipes (
    id uuid primary key default gen_random_uuid(),
    user_id uuid not null references auth.users(id) on delete cascade,
    cuisine_id uuid references recipes.cuisines(id),
    title text not null,
    description text,
    instructions text,
    servings int,
    prep_time_minutes int,
    cook_time_minutes int,
    image_path text,
    share_id text unique default encode(gen_random_bytes(6), 'hex'),
    created_at timestamptz not null default now(),
    updated_at timestamptz not null default now()
);

-- Ingredients: normalized, one row per ingredient
create table recipes.ingredients (
    id uuid primary key default gen_random_uuid(),
    recipe_id uuid not null references recipes.recipes(id) on delete cascade,
    name text not null,
    quantity numeric,
    unit text,
    position int not null default 0
);

-- Sharing: grants a user access to a recipe that isn't theirs
create table recipes.recipe_shares (
    id uuid primary key default gen_random_uuid(),
    recipe_id uuid not null references recipes.recipes(id) on delete cascade,
    shared_with_user_id uuid not null references auth.users(id) on delete cascade,
    created_at timestamptz not null default now(),
    unique (recipe_id, shared_with_user_id)
);

-- Row Level Security
alter table recipes.cuisines enable row level security;
alter table recipes.recipes enable row level security;
alter table recipes.ingredients enable row level security;
alter table recipes.recipe_shares enable row level security;

create policy "Anyone authenticated can view cuisines"
    on recipes.cuisines for select
    using (auth.role() = 'authenticated');

create policy "Owners can do everything on their recipes"
    on recipes.recipes for all
    using (auth.uid() = user_id);

create policy "Shared users can view recipes"
    on recipes.recipes for select
    using (
        exists (
            select 1 from recipes.recipe_shares
            where recipe_shares.recipe_id = recipes.recipes.id
            and recipe_shares.shared_with_user_id = auth.uid()
        )
    );

create policy "Ingredients follow recipe access"
    on recipes.ingredients for select
    using (
        exists (
            select 1 from recipes.recipes
            where recipes.recipes.id = ingredients.recipe_id
            and (recipes.recipes.user_id = auth.uid()
                 or exists (
                     select 1 from recipes.recipe_shares
                     where recipe_shares.recipe_id = recipes.recipes.id
                     and recipe_shares.shared_with_user_id = auth.uid()
                 ))
        )
    );

create policy "Owners manage their recipe's ingredients"
    on recipes.ingredients for insert
    with check (
        exists (select 1 from recipes.recipes where recipes.recipes.id = ingredients.recipe_id and recipes.recipes.user_id = auth.uid())
    );

create policy "Owners update their recipe's ingredients"
    on recipes.ingredients for update
    using (
        exists (select 1 from recipes.recipes where recipes.recipes.id = ingredients.recipe_id and recipes.recipes.user_id = auth.uid())
    );

create policy "Owners delete their recipe's ingredients"
    on recipes.ingredients for delete
    using (
        exists (select 1 from recipes.recipes where recipes.recipes.id = ingredients.recipe_id and recipes.recipes.user_id = auth.uid())
    );

create policy "Users can add themselves via share"
    on recipes.recipe_shares for insert
    with check (auth.uid() = shared_with_user_id);

create policy "Users can see their own shares"
    on recipes.recipe_shares for select
    using (auth.uid() = shared_with_user_id);
